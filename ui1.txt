local function create(className, properties, parent)
    local instance = Instance.new(className)
    for property, value in pairs(properties or {}) do
        instance[property] = value
    end
    if parent then
        instance.Parent = parent
    end
    return instance
end

local function corner(parent, radius)
    return create("UICorner", {CornerRadius = UDim.new(0, radius or 12)}, parent)
end

local function stroke(parent, color, transparency, thickness)
    return create("UIStroke", {
        Color = color or Theme.Border,
        Transparency = transparency == nil and 0.72 or transparency,
        Thickness = thickness or 1,
    }, parent)
end

local function gradient(parent, first, second, rotation, transparency)
    return create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, first),
            ColorSequenceKeypoint.new(1, second),
        }),
        Rotation = rotation or 0,
        Transparency = transparency or NumberSequence.new(0),
    }, parent)
end

local function playTween(instance, duration, properties, style)
    local info = TweenInfo.new(
        duration or 0.18,
        style or Enum.EasingStyle.Quint,
        Enum.EasingDirection.Out
    )
    local animation = TweenService:Create(instance, info, properties)
    animation:Play()
    return animation
end

local function listen(window, signal, callback)
    local connection = signal:Connect(callback)
    table.insert(window._connections, connection)
    return connection
end

local function invoke(callback, ...)
    if type(callback) ~= "function" then
        return
    end
    local ok, message = pcall(callback, ...)
    if not ok then
        warn("[GlassUI] " .. tostring(message))
    end
end

local function label(parent, text, size, color, font, xAlignment)
    return create("TextLabel", {
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Text = tostring(text or ""),
        TextColor3 = color or Theme.Text,
        TextSize = size or 13,
        Font = font or Enum.Font.Gotham,
        TextXAlignment = xAlignment or Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Center,
        TextWrapped = false,
        RichText = false,
    }, parent)
end

local function icon(parent, name, color)
    local root = create("Frame", {
        Name = "Icon",
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Size = UDim2.fromOffset(18, 18),
    }, parent)
    local parts = {}
    local ink = color or Theme.Muted

    local function addPart(object, property)
        table.insert(parts, {Object = object, Property = property})
        return object
    end

    local function line(x1, y1, x2, y2, thickness)
        local dx = x2 - x1
        local dy = y2 - y1
        local length = math.sqrt(dx * dx + dy * dy)
        local object = create("Frame", {
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromOffset((x1 + x2) * 0.5, (y1 + y2) * 0.5),
            Size = UDim2.fromOffset(length, thickness or 1.6),
            Rotation = math.deg(math.atan2(dy, dx)),
            BackgroundColor3 = ink,
            BorderSizePixel = 0,
        }, root)
        corner(object, 2)
        addPart(object, "BackgroundColor3")
        return object
    end

    local function ring(x, y, width, height, filled)
        local object = create("Frame", {
            Position = UDim2.fromOffset(x, y),
            Size = UDim2.fromOffset(width, height),
            BackgroundColor3 = ink,
            BackgroundTransparency = filled and 0 or 1,
            BorderSizePixel = 0,
        }, root)
        corner(object, math.max(width, height))
        if filled then
            addPart(object, "BackgroundColor3")
        else
            local outline = stroke(object, ink, 0, 1.5)
            addPart(outline, "Color")
        end
        return object
    end

    local key = string.lower(tostring(name or "grid"))
    if key == "home" or key == "dashboard" then
        line(2.5, 8, 9, 2.5)
        line(9, 2.5, 15.5, 8)

    elseif key == "sliders" or key == "controls" then
        line(3, 4, 15, 4)
        line(3, 9, 15, 9)
        line(3, 14, 15, 14)
        ring(5, 2, 4, 4, false)
        ring(10, 7, 4, 4, false)

            addPart(stroke(object, ink, 0, 1.4), "Color")
        end
    elseif key == "appearance" or key == "sun" then
        ring(5, 5, 8, 8, false)
        ring(8, 8, 2, 2, true)
        line(9, 1.5, 9, 3)
        line(9, 15, 9, 16.5)
        line(1.5, 9, 3, 9)
        line(15, 9, 16.5, 9)
        line(3.5, 3.5, 4.7, 4.7)
        line(13.3, 13.3, 14.5, 14.5)
        line(13.3, 4.7, 14.5, 3.5)
        line(3.5, 14.5, 4.7, 13.3)
    elseif key == "info" then
        ring(3, 3, 12, 12, false)
        line(9, 8, 9, 12.5, 1.7)
        ring(8.2, 5.4, 1.6, 1.6, true)
    elseif key == "close" or key == "cross" then
        line(4, 4, 14, 14, 1.8)
        line(14, 4, 4, 14, 1.8)
    elseif key == "chevron" or key == "down" then
        line(4.5, 7, 9, 11.5, 1.8)
        line(9, 11.5, 13.5, 7, 1.8)
    elseif key == "check" or key == "success" then
        line(3.5, 9, 7.2, 12.7, 1.8)

        line(2, 15, 9, 2, 1.5)
        line(9, 6, 9, 10.5, 1.7)
        ring(8.2, 12, 1.6, 1.6, true)
    elseif key == "bell" then
        line(5, 13.5, 13, 13.5)
        line(4.5, 12.5, 5.5, 10)
        line(5.5, 10, 5.5, 7.5)
        line(5.5, 7.5, 7, 5)
        line(7, 5, 11, 5)
        line(11, 5, 12.5, 7.5)
        line(12.5, 7.5, 12.5, 10)

local function card(parent, window, height, compact)
    local frame = create("Frame", {
        Size = UDim2.new(1, 0, 0, height or 56),
        BackgroundColor3 = Theme.Surface,
        BackgroundTransparency = compact and 0.32 or 0.18,
        BorderSizePixel = 0,
    }, parent)
    corner(frame, 13)
    local outline = stroke(frame, Theme.Border, 0.84, 1)
    frame.MouseEnter:Connect(function()
        if frame.Parent then
            playTween(frame, 0.14, {BackgroundColor3 = Theme.SurfaceRaised})
            playTween(outline, 0.14, {Transparency = 0.7})
        end
    end)
    frame.MouseLeave:Connect(function()
        if frame.Parent then
            playTween(frame, 0.16, {BackgroundColor3 = Theme.Surface})
            playTween(outline, 0.16, {Transparency = 0.84})
        end
    end)
    return frame
end

local function makeScope(window, parent)
    return setmetatable({_window = window, _parent = parent, _items = {}}, Scope)
end

local function makeButton(window, parent, config, compact)
    config = config or {}
    local title = tostring(config.Name or config.Title or "Action")
    local description = config.Description and tostring(config.Description) or nil
    local height = compact and 42 or (description and 64 or 52)
    local frame = card(parent, window, height, compact)
    local button = create("TextButton", {
        Name = "ActionButton",
        Size = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1,
        AutoButtonColor = false,
        Text = "",
        BorderSizePixel = 0,
    }, frame)

    local iconName = config.Icon or "chevron"
    local glyph = icon(button, iconName, Theme.Accent)
    glyph.Root.Position = UDim2.fromOffset(14, (height - 18) * 0.5)

    local titleLabel = label(button, title, compact and 12 or 13, Theme.Text, Enum.Font.GothamMedium)
    titleLabel.Position = UDim2.fromOffset(43, description and 9 or 0)
    titleLabel.Size = UDim2.new(1, -77, 0, description and 20 or height)

    if description then
        local detail = label(button, description, 11, Theme.Muted)
        detail.Position = UDim2.fromOffset(43, 31)
        detail.Size = UDim2.new(1, -66, 0, 19)
    end

    if string.lower(tostring(iconName)) == "chevron" then
        glyph.Root.Position = UDim2.new(1, -31, 0.5, -9)
        glyph:SetColor(Theme.Muted)
    end

    listen(window, button.Activated, function()
        invoke(config.Callback)
    end)

    local handle = {}
    function handle:Destroy()
        if frame.Parent then
            frame:Destroy()
        end
    end
    return handle
end

local function makeToggle(window, parent, config, compact)
    config = config or {}
    local title = tostring(config.Name or config.Title or "Toggle")
    local description = config.Description and tostring(config.Description) or nil
    local hasSettings = not compact and type(config.Settings) == "function"
    local headerHeight = compact and 40 or 58
    local frame = card(parent, window, hasSettings and 0 or headerHeight, compact)
    local header = create("Frame", {
        Size = UDim2.new(1, 0, 0, headerHeight),
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
    }, frame)

    if hasSettings then
        frame.Size = UDim2.new(1, 0, 0, 0)
        frame.AutomaticSize = Enum.AutomaticSize.Y
        create("UIListLayout", {
            SortOrder = Enum.SortOrder.LayoutOrder,
            Padding = UDim.new(0, 0),
        }, frame)
        header.LayoutOrder = 1
    end

    local sideSpace = hasSettings and 112 or 70
    local titleLabel = label(header, title, compact and 12 or 13, Theme.Text, Enum.Font.GothamMedium)
    titleLabel.Position = UDim2.fromOffset(14, description and 7 or 0)
    titleLabel.Size = UDim2.new(1, -sideSpace, 0, description and 21 or headerHeight)

    if description and not compact then
        local detail = label(header, description, 11, Theme.Muted)
        detail.Position = UDim2.fromOffset(14, 31)
        detail.Size = UDim2.new(1, -sideSpace, 0, 18)
    end

    local switch = create("TextButton", {
        Name = "Switch",
        AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, -14, 0.5, 0),
        Size = UDim2.fromOffset(compact and 36 or 42, compact and 20 or 22),
        BackgroundColor3 = Theme.SurfaceRaised,
        BorderSizePixel = 0,
        AutoButtonColor = false,
        Text = "",
    }, header)
    corner(switch, 20)

    local knobSize = compact and 14 or 16
    local knob = create("Frame", {
        AnchorPoint = Vector2.new(0, 0.5),
        Position = UDim2.new(0, 3, 0.5, 0),
        Size = UDim2.fromOffset(knobSize, knobSize),
        BackgroundColor3 = Theme.Text,
        BorderSizePixel = 0,
    }, switch)
    corner(knob, 20)
    stroke(knob, Theme.Border, 0.72, 0.8)

    local value = config.Default == true
    local function setValue(nextValue, fire)
        value = nextValue == true
        playTween(switch, 0.16, {
            BackgroundColor3 = value and Theme.Accent or Theme.SurfaceRaised,
        })
        playTween(knob, 0.16, {
            Position = value
                and UDim2.new(1, -knobSize - 3, 0.5, 0)
                or UDim2.new(0, 3, 0.5, 0),
        })
        if fire then
            invoke(config.Callback, value)
        end
    end
    setValue(value, false)

    listen(window, switch.Activated, function()
        setValue(not value, true)
    end)

    local handle = {}
    function handle:SetValue(nextValue, fire)
        setValue(nextValue, fire == true)
    end
    function handle:GetValue()
        return value
    end

    if hasSettings then
        local expandButton = create("TextButton", {
            Name = "ExpandSettings",
            AnchorPoint = Vector2.new(1, 0.5),
            Position = UDim2.new(1, -60, 0.5, 0),
            Size = UDim2.fromOffset(30, 30),
            BackgroundTransparency = 1,
            AutoButtonColor = false,
            Text = "",
        }, header)
        local arrow = icon(expandButton, "chevron", Theme.Muted)
        arrow.Root.Position = UDim2.fromOffset(6, 6)

        local clip = create("Frame", {
            Name = "InlineSettings",
            Size = UDim2.new(1, 0, 0, 0),
            BackgroundTransparency = 1,
            BorderSizePixel = 0,
            ClipsDescendants = true,
            Visible = false,
            LayoutOrder = 2,
        }, frame)
        local inner = create("Frame", {
            Position = UDim2.fromOffset(12, 0),
            Size = UDim2.new(1, -24, 0, 0),
            AutomaticSize = Enum.AutomaticSize.Y,
            BackgroundTransparency = 1,
            BorderSizePixel = 0,
        }, clip)
        create("UIPadding", {
            PaddingTop = UDim.new(0, 8),
            PaddingBottom = UDim.new(0, 12),
            PaddingLeft = UDim.new(0, 0),
            PaddingRight = UDim.new(0, 0),
        }, inner)
        local settingsLayout = create("UIListLayout", {
            SortOrder = Enum.SortOrder.LayoutOrder,
            Padding = UDim.new(0, 6),
        }, inner)
        local scope = makeScope(window, inner)
        invoke(config.Settings, scope)

        if #scope._items == 0 then
            expandButton.Visible = false
            titleLabel.Size = UDim2.new(1, -70, 0, description and 21 or headerHeight)
            if description then
                local detail = header:FindFirstChildWhichIsA("TextLabel")
                if detail and detail ~= titleLabel then
                    detail.Size = UDim2.new(1, -70, 0, 18)
                end
            end
            frame.AutomaticSize = Enum.AutomaticSize.None
            frame.Size = UDim2.new(1, 0, 0, headerHeight)
        else
            local expanded = false
            local revision = 0
            local function targetHeight()
                return settingsLayout.AbsoluteContentSize.Y + 20
            end
            local function refreshHeight()
                if expanded and clip.Parent then
                    playTween(clip, 0.12, {Size = UDim2.new(1, 0, 0, targetHeight())})
                end
            end
            listen(window, settingsLayout:GetPropertyChangedSignal("AbsoluteContentSize"), refreshHeight)

            local function setExpanded(nextState)
                expanded = nextState
                revision = revision + 1
                local currentRevision = revision
                arrow.Root.Rotation = expanded and 180 or 0
                if expanded then
                    clip.Visible = true
                    playTween(clip, 0.2, {Size = UDim2.new(1, 0, 0, targetHeight())})
                else
                    playTween(clip, 0.16, {Size = UDim2.new(1, 0, 0, 0)})
                    task.delay(0.18, function()
                        if not expanded and revision == currentRevision and clip.Parent then
                            clip.Visible = false
                        end
                    end)
                end
            end
            listen(window, expandButton.Activated, function()
                setExpanded(not expanded)
            end)
        end
    end

    return handle
end

local function formatNumber(value, step)
    if step >= 1 then
        return tostring(math.floor(value + 0.5))
    end
    local text = tostring(step)
    local decimal = string.find(text, "%.")
    local places = decimal and math.min(4, #text - decimal) or 2
    return string.format("%." .. tostring(places) .. "f", value)
end

local function makeSlider(window, parent, config, compact)
    config = config or {}
    local title = tostring(config.Name or config.Title or "Value")
    local minimum = tonumber(config.Min) or 0
    local maximum = tonumber(config.Max) or 100
    if maximum <= minimum then
        maximum = minimum + 1
    end
    local step = tonumber(config.Step) or (maximum - minimum) / 100
    if step <= 0 then
        step = (maximum - minimum) / 100
    end
    local height = compact and 58 or 76
    local frame = card(parent, window, height, compact)
    local titleLabel = label(frame, title, compact and 11 or 12, Theme.Text, Enum.Font.GothamMedium)
    titleLabel.Position = UDim2.fromOffset(14, 5)
    titleLabel.Size = UDim2.new(1, -92, 0, 20)

    local valueLabel = label(frame, "", 11, Theme.Accent, Enum.Font.GothamMedium, Enum.TextXAlignment.Right)
    valueLabel.Position = UDim2.new(1, -72, 0, 5)
    valueLabel.Size = UDim2.fromOffset(58, 20)

    local track = create("TextButton", {
        Name = "SliderTrack",
        Position = UDim2.new(0, 14, 0, compact and 37 or 48),
        Size = UDim2.new(1, -28, 0, 6),
        BackgroundColor3 = Theme.SurfaceRaised,
        BorderSizePixel = 0,
        AutoButtonColor = false,
        Text = "",
    }, frame)
    corner(track, 8)

    local fill = create("Frame", {
        Size = UDim2.new(0, 0, 1, 0),
        BackgroundColor3 = Theme.Accent,
        BorderSizePixel = 0,
    }, track)
    corner(fill, 8)
    gradient(fill, Theme.Accent, Theme.AccentAlt, 0)

    local knob = create("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0, 0, 0.5, 0),
        Size = UDim2.fromOffset(14, 14),
        BackgroundColor3 = Theme.Text,
        BorderSizePixel = 0,
    }, track)
    corner(knob, 20)
    stroke(knob, Theme.Accent, 0.14, 1.2)

    local value = tonumber(config.Default) or minimum
    local function updateVisual(nextValue, fire)
        local snapped = minimum + math.floor((nextValue - minimum) / step + 0.5) * step
        value = math.clamp(snapped, minimum, maximum)
        local alpha = math.clamp((value - minimum) / (maximum - minimum), 0, 1)
        fill.Size = UDim2.new(alpha, 0, 1, 0)
        knob.Position = UDim2.new(alpha, 0, 0.5, 0)
        valueLabel.Text = formatNumber(value, step)
        if fire then
            invoke(config.Callback, value)
        end
    end
    updateVisual(value, false)

    local dragging = false
    local function updateFromX(x, fire)
        local width = math.max(track.AbsoluteSize.X, 1)
        local alpha = math.clamp((x - track.AbsolutePosition.X) / width, 0, 1)
        updateVisual(minimum + alpha * (maximum - minimum), fire)
    end

    listen(window, track.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            updateFromX(input.Position.X, true)
        end
    end)
    listen(window, UserInputService.InputChanged, function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateFromX(input.Position.X, true)
        end
    end)
    listen(window, UserInputService.InputEnded, function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
            dragging = false
        end
    end)

    local handle = {}
    function handle:SetValue(nextValue, fire)
        updateVisual(tonumber(nextValue) or minimum, fire == true)
    end
    function handle:GetValue()
        return value
    end
    return handle
end

            TextColor3 = Theme.Muted,
            TextSize = 11,
            Font = Enum.Font.Gotham,
            TextXAlignment = Enum.TextXAlignment.Left,
            LayoutOrder = index,
        }, scroll)
        corner(optionButton, 8)
        create("UIPadding", {PaddingLeft = UDim.new(0, 10)}, optionButton)
        listen(window, optionButton.Activated, function()
            setValue(option, true)
            expanded = false
            arrow.Root.Rotation = 0
            playTween(clip, 0.16, {Size = UDim2.new(1, 0, 0, 0)})
            task.delay(0.18, function()
                if not expanded and clip.Parent then
                    clip.Visible = false
                end
            end)
        end)
        listen(window, optionButton.MouseEnter, function()
            optionButton.BackgroundTransparency = 0.14
            optionButton.TextColor3 = Theme.Text
        end)
        listen(window, optionButton.MouseLeave, function()
            optionButton.BackgroundTransparency = 0.5
            optionButton.TextColor3 = Theme.Muted
        end)
    end

    local function setExpanded(nextState)
        if optionCount == 0 then
            return
        end
        expanded = nextState
        arrow.Root.Rotation = expanded and 180 or 0
        if expanded then
            local visibleHeight = math.min(optionCount * 33, 132) + 8
            scroll.Size = UDim2.new(1, -16, 0, visibleHeight - 8)
            clip.Visible = true
            playTween(clip, 0.18, {Size = UDim2.new(1, 0, 0, visibleHeight)})
        else
            playTween(clip, 0.16, {Size = UDim2.new(1, 0, 0, 0)})
            task.delay(0.18, function()
                if not expanded and clip.Parent then
                    clip.Visible = false
                end
            end)
        end
    end

    listen(window, header.Activated, function()
        setExpanded(not expanded)
    end)

    local handle = {}
    function handle:SetValue(nextValue, fire)
        setValue(nextValue, fire == true)
    end
    function handle:GetValue()
        return current
    end
    function handle:ToggleOptions()
        setExpanded(not expanded)
    end
    return handle
end

local function selectTab(window, entry)
    if window._destroyed or not entry then
        return
    end
    window._activeTab = entry
    for _, item in ipairs(window._tabs) do
        local selected = item == entry
        item.Page.Visible = selected
        item.Button.BackgroundColor3 = selected and Theme.SurfaceRaised or Theme.Surface
        item.Button.BackgroundTransparency = selected and 0.18 or 1
        item.Stroke.Transparency = selected and 0.76 or 1
        item.Icon:SetColor(selected and Theme.Accent or Theme.Muted)
        item.Label.TextColor3 = selected and Theme.Text or Theme.Muted
    end
    window._pageTitle.Text = entry.Name
    window._pageDescription.Text = entry.Description or "Interface controls"
end

local function closeToast(window, record)
    if record.Closed then
        return
    end
    record.Closed = true
    for index, item in ipairs(window._toasts) do
        if item == record then
            table.remove(window._toasts, index)
            break
        end
    end
    if window._destroyed or not record.Outer.Parent then
        return
    end
    if record.ProgressTween then
        record.ProgressTween:Cancel()
    end
    playTween(record.Surface, 0.16, {BackgroundTransparency = 1})
    playTween(record.Outer, 0.18, {Size = UDim2.new(1, 0, 0, 0)})
    playTween(record.Title, 0.12, {TextTransparency = 1})
    playTween(record.Content, 0.12, {TextTransparency = 1})
    task.delay(0.2, function()
        if record.Outer.Parent then
            record.Outer:Destroy()
        end
    end)
end

function Window:Notify(config)
    if self._destroyed then
        return
    end
    config = config or {}
    local kind = string.lower(tostring(config.Type or config.Kind or "info"))
    local tones = {
        info = {Color = Theme.Accent, Icon = "info"},
        success = {Color = Theme.Success, Icon = "check"},
        warning = {Color = Theme.Warning, Icon = "warning"},
        error = {Color = Theme.Error, Icon = "close"},
    }
    local tone = tones[kind] or tones.info
    local duration = math.clamp(tonumber(config.Duration) or 4.5, 1, 20)
    local titleText = tostring(config.Title or "Notification")
    local contentText = tostring(config.Content or config.Message or "")

    if #self._toasts >= 4 then
        closeToast(self, self._toasts[1])
    end
    self._toastSequence = self._toastSequence + 1

    local outer = create("Frame", {
        Name = "Toast",
        Size = UDim2.new(1, 0, 0, 0),
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        ClipsDescendants = true,
        LayoutOrder = -self._toastSequence,
    }, self._toastStack)
    local surface = create("Frame", {
        Size = UDim2.new(1, 0, 0, 76),
        BackgroundColor3 = Theme.Surface,
        BackgroundTransparency = 0.16,
        BorderSizePixel = 0,
    }, outer)
    corner(surface, 13)
    stroke(surface, Theme.Border, 0.76, 1)
    gradient(surface, Theme.SurfaceRaised, Theme.Surface, 12, NumberSequence.new(0.08))

    local accentBar = create("Frame", {
        Position = UDim2.fromOffset(0, 14),
        Size = UDim2.new(0, 3, 0, 48),
        BackgroundColor3 = tone.Color,
        BorderSizePixel = 0,
    }, surface)
    corner(accentBar, 3)
    local glyph = icon(surface, tone.Icon, tone.Color)
    glyph.Root.Position = UDim2.fromOffset(14, 16)

    local titleLabel = label(surface, titleText, 12, Theme.Text, Enum.Font.GothamMedium)
    titleLabel.Position = UDim2.fromOffset(42, 9)
    titleLabel.Size = UDim2.new(1, -78, 0, 19)
    local contentLabel = label(surface, contentText, 11, Theme.Muted)
    contentLabel.Position = UDim2.fromOffset(42, 30)
    contentLabel.Size = UDim2.new(1, -54, 0, 34)
    contentLabel.TextWrapped = true
    contentLabel.TextYAlignment = Enum.TextYAlignment.Top

    local dismiss = create("TextButton", {
        Name = "Dismiss",
        AnchorPoint = Vector2.new(1, 0),
        Position = UDim2.new(1, -8, 0, 7),
        Size = UDim2.fromOffset(22, 22),
        BackgroundTransparency = 1,
        AutoButtonColor = false,
        Text = "",
    }, surface)
    local closeGlyph = icon(dismiss, "close", Theme.Quiet)
    closeGlyph.Root.Position = UDim2.fromOffset(2, 2)

    local progressTrack = create("Frame", {
        Position = UDim2.new(0, 12, 1, -4),
        Size = UDim2.new(1, -24, 0, 2),
        BackgroundColor3 = Theme.Border,
        BackgroundTransparency = 0.87,
        BorderSizePixel = 0,
    }, surface)
    corner(progressTrack, 2)
    local progress = create("Frame", {
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = tone.Color,
        BorderSizePixel = 0,
    }, progressTrack)
    corner(progress, 2)

    local record = {
        Outer = outer,
        Surface = surface,
        Title = titleLabel,
        Content = contentLabel,
        Progress = progress,
        Closed = false,
    }
    table.insert(self._toasts, record)
    listen(self, dismiss.Activated, function()
        closeToast(self, record)
    end)

    playTween(outer, 0.22, {Size = UDim2.new(1, 0, 0, 76)})
    playTween(surface, 0.22, {BackgroundTransparency = 0.12})
    record.ProgressTween = playTween(progress, duration, {Size = UDim2.new(0, 0, 1, 0)}, Enum.EasingStyle.Linear)
    task.delay(duration, function()
        closeToast(self, record)
    end)
    return record
end

function Window:Destroy()
    if self._destroyed then
        return
    end
    self._destroyed = true
    for _, connection in ipairs(self._connections) do
        if connection.Connected then
            connection:Disconnect()
        end
    end
    table.clear(self._connections)
    if self.Gui and self.Gui.Parent then
        self.Gui:Destroy()
    end
end

        listen(self, UserInputService.InputBegan, function(input, gameProcessed)
            if gameProcessed or UserInputService:GetFocusedTextBox() then
                return
            end
            if input.KeyCode == toggleKey then
                self:Toggle()
            end
        end)
    end

    return self
end

return Library
